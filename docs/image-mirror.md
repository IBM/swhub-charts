2. Copy/Mirror image to private registry
```bash
skopeo login cp.icr.io -u cp -p <Entitlement key>
skopeo login <private registry> -u <private_registry_username> -p <private_registry_password>

IMAGE_LIST=./list_images.txt
PRIVATE_REGISTRY=<Private registry>

for image in $(cat "$IMAGE_LIST")
do
    SRC_IMG="${image}"
    TRG_IMG="${PRIVATE_REGISTRY}/${image#*/}"
    echo "[INFO] Copying  docker://${SRC_IMG}  to  docker://${TRG_IMG}"
    CMD="skopeo copy --src-tls-verify=false --dest-tls-verify=false --preserve-digests --all docker://${SRC_IMG} docker://${TRG_IMG}"
    $CMD 
done
```

3. Verify all images are mirrored
```bash
IMAGE_LIST=./list_images.txt
PRIVATE_REGISTRY=<private registry>
OUTPUT_FILE=./mirror_result.txt
echo "" > $OUTPUT_FILE

for image in $(cat "$IMAGE_LIST")
do
    TRG_IMG="${PRIVATE_REGISTRY}/${image#*/}"
    CMD="skopeo inspect --raw --override-os linux  --tls-verify=false docker://${TRG_IMG}"
    $CMD > /dev/null 2>&1
    if [ $? -ne 0 ]; then
        echo "FAILED - $CMD" >> $OUTPUT_FILE
    else
        echo "OK - $TRG_IMG" >> $OUTPUT_FILE
    fi
done
```
